/* ── Message rendering ──────────────────────────────────────── */
function escapeHtml(str) {
  return str.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
            .replace(/"/g,'&quot;').replace(/'/g,'&#039;');
}

function formatMsgText(text) {
  /* Minimal markdown-like formatting for AI responses */
  return text
    .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
    .replace(/`(.+?)`/g, '<code>$1</code>')
    .replace(/\n\n/g, '</p><p>')
    .replace(/\n/g, '<br>');
}

/* ── Main send flow ─────────────────────────────────────────── */
async function sendMessage() {
  const text = el.input.value.trim();
  if (!text || state.isTyping) return;

  /* Reset input */
  el.input.value = '';
  el.input.style.height = 'auto';
  el.sendBtn.disabled = true;
  el.sendBtn.classList.remove('active');
  el.charCount.textContent = '';

  /* Update topbar title with first message */
  if (!state.activeConvId) {
    const short = text.length > 48 ? text.slice(0, 45) + '…' : text;
    el.topbarTitle.innerHTML = `<span>${escapeHtml(short)}</span>`;
    /* Add to conversation list */
    const newConv = { id: 'new-' + Date.now(), title: short, date: 'Maintenant' };
    state.conversations.unshift(newConv);
    state.activeConvId = newConv.id;
    renderConvList();
  }

  state.isTyping = true;
  appendUserMessage(text);
  showTypingIndicator();

  /* ── BACKEND CALL (replace this block with real GAS endpoint) ── */
  /*
    To connect to your Google Apps Script backend:

    const response = await fetch('YOUR_GAS_DEPLOYMENT_URL', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ message: text, conversationId: state.activeConvId })
    });
    const data = await response.json();
    const aiText = data.reply;
    const sources = data.sources || [];
    const areas   = data.areas   || [];
  */

  /* ── Demo simulation (remove when connecting backend) ── */
  await simulatedDelay(1200 + Math.random() * 800);
  const { aiText, sources, areas } = getDemoResponse(text);
  /* ── End demo simulation ── */

  removeTypingIndicator();
  appendAiMessage(aiText, sources);

  /* Append Area Explorer if the response includes related domains */
  if (areas && areas.length) {
    const lastAiGroup = el.chatInner.querySelector('.message-group.ai:last-child');
    if (lastAiGroup) {
      const aiContent = lastAiGroup.querySelector('.ai-content');
      if (aiContent) aiContent.appendChild(buildAreaExplorer(areas));
    }
  }

  state.isTyping = false;
  el.input.focus();
}

/* ── Responsive: handle window resize ─────────────────────── */
let resizeDebounce;
window.addEventListener('resize', () => {
  clearTimeout(resizeDebounce);
  resizeDebounce = setTimeout(() => {
    if (window.innerWidth > 768) {
      el.sidebar.classList.remove('mobile-open');
      el.overlay.classList.remove('visible');
      el.overlay.setAttribute('aria-hidden', 'true');
      if (!state.sidebarOpen) el.sidebar.classList.add('collapsed');
    }
  }, 100);
});

/* ── Init focus ─────────────────────────────────────────────── */
window.addEventListener('load', () => {
  el.input.focus();
  /* Restore initial sidebar state based on viewport */
  if (window.innerWidth <= 768) {
    setSidebar(false);
  }
});

/* ── Keyboard accessibility: Tab trap in modal-like elements ─ */
document.addEventListener('keydown', e => {
  /* Alt+N: new conversation */
  if (e.altKey && e.key === 'n') { e.preventDefault(); el.newChat.click(); }
  /* Alt+D: toggle dark mode */
  if (e.altKey && e.key === 'd') { e.preventDefault(); el.themeToggle.click(); }
  /* / focuses input */
  if (e.key === '/' && document.activeElement !== el.input) {
    e.preventDefault(); el.input.focus();
  }
});
